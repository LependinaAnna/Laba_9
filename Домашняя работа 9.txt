Условие задачи 16. Написать программу, выводящую на экран заданную геометрическую фигуру, нарисованную с помощью заданного с клавиатуры символа, условия заполнения фигуры и задаваемые параметры фигуры смотреть ниже (остальные константы). Предусмотреть возможность изменения размеров фигуры путем задания её размеров в количестве символов пользователем.
Фигура: четырёхугольник (описанный около окружности радиусом r); заполнение контура: нет (контур); задаваемые пользователем параметры: r (радиус вписанной окружности).

1. Алгоритм.
Начало 
Определить и ввести константы:
1) r - радиус вписанной окружности
2) symbol - какой-либо символ
3) contin - какое-либо число
Учитывая введённые данные, построить фигуру (четырёхугольник).
Пока contin не станет равным нулю, продолжать программу, т. е. строить новые черырёхугольники.
Если contin равен нулю, завершить программу.
Вывести результат.
Конец

3. Реализация программы.
#define _CRT_SECURE_NO_DEPRECATE
#define _USE_MATH_DEFINES 
#include <locale.h>
#include <stdio.h>
#include <math.h>
#include <string.h>
#include <ctype.h>

int homework(int i, int j, int r);

int main()
{
    setlocale(LC_ALL, "RUS");
    
    char symbol;
    int r, contin;

    do 
    {
        puts("Введите символ, из которого будет строиться фигура");
        scanf(" %c", &symbol); 
        puts("Введите радиус вписанной окружности");
        scanf("%d", &r);
        if (r <= 0) 
        {
            printf("Ошибка! Радиус должен быть больше 0\n");
        } 
        else 
        {
            int side, result;
            side = 2 * r; 
            printf("Четырёхугольник пустой внутри (контур из '%c' четырёхугольника со стороной %d)\n\n",symbol, side);
            for (int i = 0; i < side; i++) 
            {
                for (int j = 0; j < side; j++) 
                {
                    result = homework(i, j, r);
                    if (result) 
                    {
                        printf(" %c", symbol);
                    } 
                    else 
                    {
                        printf("  ");
                    }
                }
                printf("\n");
            }
        }
        puts("\nПродолжить? (Да – введите любое число, Нет – введите 0)\n");
        scanf("%d", &contin);
    } 
    while (contin != 0);
    printf("Программа завершена\n");
    return 0;
}

int homework(int i, int j, int r)
{
    int res, side;
    side = 2 * r;
    res = i == 0 || i == side - 1 || j == 0 || j == side - 1;
    return res;
}
